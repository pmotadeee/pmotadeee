```python
        """Scan completo com aprendizado adaptativo"""
        results = {}
        for port in ports:
            open = self.scan_port(host, port)
            results[port] = open
            
            # Adaptação baseada em respostas
            if open and self.adaptive:
                self.jitter_min = max(0.5, self.jitter_min * 0.8)
        return results
```
