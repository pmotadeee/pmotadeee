function distribuir() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Dados");
  // Supondo que você tenha uma tabela com destinos e EMAs em alguma aba
  var dados = sheet.getRange("A2:B10").getValues(); // coluna A: destino, B: EMA
  var numChutes = 8;
  
  // Ordenar por EMA decrescente
  dados.sort(function(a, b) { return b[1] - a[1]; });
  
  // Inicializar chutes
  var chutes = [];
  for (var i = 0; i < numChutes; i++) {
    chutes.push({id: i+1, carga: 0, destinos: []});
  }
  
  // Distribuir
  for (var d = 0; d < dados.length; d++) {
    var destino = dados[d][0];
    var ema = dados[d][1];
    
    // Encontrar chute com menor carga
    var menorCarga = chutes[0].carga;
    var idxMenor = 0;
    for (var c = 1; c < chutes.length; c++) {
      if (chutes[c].carga < menorCarga) {
        menorCarga = chutes[c].carga;
        idxMenor = c;
      }
    }
    
    chutes[idxMenor].destinos.push(destino);
    chutes[idxMenor].carga += ema;
  }
  
  // Exibir resultado (pode jogar em outra planilha)
  var resultadoSheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Resultado");
  resultadoSheet.clear();
  resultadoSheet.appendRow(["Chute", "Destinos", "Carga Total"]);
  for (var c = 0; c < chutes.length; c++) {
    resultadoSheet.appendRow([chutes[c].id, chutes[c].destinos.join(", "), chutes[c].carga]);
  }
}

  var dados = sheet.getRange("A2:H30").getValues(); // Ajuste o intervalo
  var numChutes = 8;

  // Extrair destinos e médias (coluna G, por exemplo)
  var destinos = [];
  for (var i = 0; i < dados.length; i++) {
    if (dados[i][0] != "") { // Se há destino
      destinos.push({
        nome: dados[i][0],
        media: dados[i][6] // coluna da média (ajuste)
      });
    }
  }

  // Ordenar por média decrescente
  destinos.sort(function(a, b) { return b.media - a.media; });

  // Inicializar chutes
  var chutes = [];
  for (var j = 0; j < numChutes; j++) {
    chutes.push({ id: j+1, carga: 0, destinos: [] });
  }

  // Distribuir round-robin ponderado
  destinos.forEach(function(d) {
    // Encontrar chute com menor carga
    var chuteMenor = chutes.reduce(function(menor, atual) {
      return atual.carga < menor.carga ? atual : menor;
    });
    chuteMenor.destinos.push(d.nome);
    chuteMenor.carga += d.media;
  });

  // Exibir resultado (por exemplo, na coluna H)
  for (var k = 0; k < chutes.length; k++) {
    sheet.getRange(k+2, 8).setValue("Chute " + chutes[k].id + ": " + chutes[k].destinos.join(", "));
  }
}
