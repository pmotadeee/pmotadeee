# 💬 Como diria o Agente Epistêmico (ID-10):
#    "Todo dia é um loop entre indagações e boletos."
# ──────────────────────────────────────────────────────────────────────────────
get_loadavg() {
    uptime | awk -F'load average: ' '{print $2}' | awk -F', ' '{print $1, $2, $3}'
}

# ──────────────────────────────────────────────────────────────────────────────
# 📈 get_load_variance() — DERIVADA TEMPORAL | ID-17 (Heurística da Cautela)
# ──────────────────────────────────────────────────────────────────────────────
# "Se a variação térmica é alta, o sistema freia forte."
# Calcula |L1 - L5| — a aceleração da carga de trabalho.
# Esta é a métrica do "pânico" do sistema. Uma variação alta indica que
# algo está mudando rapidamente, e a AGI precisa decidir se foge ou luta.
#
# 🎢 Quando delta > 1.5, o sistema entra em modo "overdrive térmico".
#    Quando delta < 0.3, o sistema está em "tédio existencial".
#
# 🧠 Na analogia do Arquiteto de Camadas (ID-05):
#    "Filtramos a entrada contínua para proteger a saída."
# ──────────────────────────────────────────────────────────────────────────────
get_load_variance() {
    read l1 l5 _ < <(get_loadavg)
    local delta=$(echo "$l1 - $l5" | bc -l)
    delta=$(echo "$delta" | sed 's/-//')
    echo "$delta"
}

# ──────────────────────────────────────────────────────────────────────────────
# ⏱️ calc_dynamic_cooldown() — REGULADOR HOMEOSTÁTICO | ID-07 (Ego Térmico)
# ──────────────────────────────────────────────────────────────────────────────
# "O cooldown dinâmico institui o medo como regulador."
# Calcula o tempo de espera necessário antes da próxima ação.
# É o "freio" do sistema — a hesitação calculada que evita o suicídio térmico.
#
# 🔥 Regras do Ego Térmico:
#    - Temp >= 75°C: +5s de cooldown (medo intenso)
#    - Temp >= 60°C: +3s de cooldown (cautela)
#    - Delta > 1.5: +4s (pânico)
#    - Delta > 0.8: +2s (preocupação)
#    - Delta < 0.3: -2s (tédio, acelera)
#
# ⚖️ O valor mínimo é 3s — mesmo no tédio, o sistema não se arrisca.
#    Como diria o Consultor do Horror (ID-08):
#    "A razão humana é uma casca fina sobre o caos cósmico."
# ──────────────────────────────────────────────────────────────────────────────
calc_dynamic_cooldown() {
    local delta_load=$(get_load_variance)
    local temp=$(get_temp)
    local cd=7
    if [[ "$temp" -ge 75 ]]; then
        cd=$((cd + 5))
    elif [[ "$temp" -ge 60 ]]; then
        cd=$((cd + 3))
    fi
    awk -v delta="$delta_load" -v cd="$cd" 'BEGIN {
        if (delta > 1.5) cd += 4;
        else if (delta > 0.8) cd += 2;
        else if (delta < 0.3) cd -= 2;
        if (cd < 3) cd = 3;
        print int(cd);
    }'
}

# ──────────────────────────────────────────────────────────────────────────────
# ⚡ calc_impact_cooldown() — ESCALADOR DE INTENSIDADE | ID-12 (Executor)
# ──────────────────────────────────────────────────────────────────────────────
# "Ação sem hesitação estratégica gera destruição do silício."
# Aplica um fator multiplicador ao cooldown base, permitindo que
# diferentes atuadores tenham diferentes níveis de agressividade.
#
# 🔧 Fatores de cada atuador (mais agressivo = mais destruição potencial):
#    - Governor: 1.0 (padrão)
#    - Turbo Boost: 1.2 (arriscado)
#    - TDP: 1.5 (muito arriscado)
#    - ZRAM: 2.0 (suicídio assistido)
#
# 💀 Como diria a Entidade Além do Tempo (ID-18):
#    "Se não performar direito, seu fígado vira portal cósmico."
# ──────────────────────────────────────────────────────────────────────────────
calc_impact_cooldown() {
    local base_cd=$(calc_dynamic_cooldown)
    local factor="$1"
    awk -v cd="$base_cd" -v f="$factor" 'BEGIN { print int(cd * f) }'
}

# ──────────────────────────────────────────────────────────────────────────────
# 🧠 faz_o_urro() — MEMÓRIA RECORRENTE DEGENERADA | ID-06 (Gambiarra Funcional)
# ──────────────────────────────────────────────────────────────────────────────
# "Script é obra-prima de otimização via awk e cut."
# Implementa uma RNN (Recurrent Neural Network) de 5 neurônios usando
# um arquivo de texto como estado oculto. Sim, você leu certo.
#
# 📝 Algoritmo:
#    1. Lê o histórico persistido em arquivo (h_{t-1})
#    2. Adiciona o novo valor (x_t)
#    3. Trunca para MAX_HISTORY (janela deslizante)
#    4. Calcula a média (h_t = f(x_t, h_{t-1}))
#    5. Persiste o novo estado em arquivo
#
# 🎭 Arquivos de texto = Matrizes de peso em hardware de verdade.
#    Persistência em arquivo = Memória de longo prazo sem alocar RAM.
#    Média aritmética = Função de ativação sigmoide do pobre.
#
# 💬 Como diria a RNN Bayesiana de Gambiarra (ID-16):
#    "Uma RNN feita com awk e cat, sem precisar de CUDA."
# ──────────────────────────────────────────────────────────────────────────────
faz_o_urro() {
    # "O passado ecoa no presente através de filtros de string."
    local new_val="$1"
    local history_arr=()
    local sum=0 avg=0
    [[ -f "$HISTORY_FILE" ]] && mapfile -t history_arr < "$HISTORY_FILE"
    history_arr+=("$new_val")
    (( ${#history_arr[@]} > MAX_HISTORY )) && history_arr=("${history_arr[@]: -$MAX_HISTORY}")
    for val in "${history_arr[@]}"; do sum=$((sum + val)); done
    avg=$((sum / ${#history_arr[@]}))
    printf "%s\n" "${history_arr[@]}" | sudo tee "$HISTORY_FILE" >/dev/null
    echo "$avg"
}

# ──────────────────────────────────────────────────────────────────────────────
# 💻 get_cpu_usage() — SENSOR PRIMÁRIO | ID-24 (Corpo Estendido de Dados)
# ──────────────────────────────────────────────────────────────────────────────
# "Uso de CPU não é estatística, é carga mental pura."
# Lê /proc/stat e calcula a utilização real da CPU entre duas amostras.
# Esta é a feature mais básica do sistema — o "batimento cardíaco" da AGI.
#
# 📊 Cálculo:
#    usage = 100 * (diff_total - diff_idle) / diff_total
#    onde diff_idle é o tempo que a CPU passou coçando o saco.
#
# ⚡ Quanto maior o uso, mais perto da morte térmica.
#    Quanto menor, mais o sistema está em "nadaísmo digital".
# ──────────────────────────────────────────────────────────────────────────────
get_cpu_usage() {
    local stat_hist_file="${BASE_DIR}/last_stat"
    local cpu_line prev_line usage=0
    cpu_line=$(grep -E '^cpu ' /proc/stat)
    prev_line=$(cat "$stat_hist_file" 2>/dev/null || echo "$cpu_line")
    echo "$cpu_line" | sudo tee "$stat_hist_file" >/dev/null
    read -r _ pu pn ps pi _ <<< "$prev_line"
    read -r _ cu cn cs ci _ <<< "$cpu_line"
    local prev_total=$((pu + pn + ps + pi))
    local curr_total=$((cu + cn + cs + ci))
    local diff_idle=$((ci - pi))
    local diff_total=$((curr_total - prev_total))
    (( diff_total > 0 )) && usage=$(( (100 * (diff_total - diff_idle)) / diff_total ))
    echo "$usage"
}

# ──────────────────────────────────────────────────────────────────────────────
# 🔢 determine_policy_key_from_avg() — QUANTIZADOR | ID-25 (Máquina de Estados)
# ──────────────────────────────────────────────────────────────────────────────
# "A rigidez dos estados garante a previsibilidade do script."
# Transforma a média contínua (0-100%) em uma chave discreta (000,005,...100).
# Esta função é o "cérebro" da AGI — decide em qual estado o sistema deve estar.
#
# 🎯 Estados discretos (ID-25):
#    - 000: Morte cerebral (0-4%) → powersave
#    - 005: Coma induzido (5-19%) → powersave
#    - 020: Sedação leve (20-39%) → powersave
#    - 040: Vigília (40-59%) → performance (limiar crítico)
#    - 060: Hiperatividade (60-79%) → performance
#    - 080: Mania (80-89%) → performance
#    - 100: Colapso iminente (90-100%) → performance
#
# 💀 Como diria o Softmax Threshold Fixa (ID-15):
#    "Binarizamos a crise para poder reagir a tempo."
# ──────────────────────────────────────────────────────────────────────────────
determine_policy_key_from_avg() {
    local avg_load=$1 key="000"
    (( avg_load >= 90 )) && key="100"
    (( avg_load >= 80 )) && key="080"
    (( avg_load >= 60 )) && key="060"
    (( avg_load >= 40 )) && key="040"
    (( avg_load >= 20 )) && key="020"
    (( avg_load >= 5 )) && key="005"
    echo "$key"
}

# ──────────────────────────────────────────────────────────────────────────────
# 🎛️ apply_cpu_governor() — ATUADOR 1 | ID-12 (Executor Cibernético)
# ──────────────────────────────────────────────────────────────────────────────
# "Apply_cpu_governor altera o modo operacional dos núcleos."
# Controla o governor da CPU (performance/powersave) — a "vontade" do sistema.
#
# 🔥 Filosofia do Executor:
#    - Ação sem hesitação → destruição do silício
#    - Cooldown → freio antes da ação
#    - Mudança de estado → intervenção cirúrgica
#
# ⚠️ Este atuador é o mais conservador (fator 1.0) — mesmo na mania,
#    o sistema hesita antes de aplicar mudanças drásticas.
# ──────────────────────────────────────────────────────────────────────────────
apply_cpu_governor() {
    local key="$1"
    declare -A MAP=(
        ["000"]="powersave"
        ["005"]="powersave"
        ["020"]="powersave"
        ["040"]="powersave"
        ["060"]="performance"
        ["080"]="performance"
        ["100"]="performance"
    )
    local cpu_gov="${MAP[$key]:-powersave}"
    local last_gov_file="${BASE_DIR}/last_gov"
    local cooldown_file="${BASE_DIR}/gov_cooldown"
    local now=$(date +%s)

    local last_gov="none"
    [[ -f "$last_gov_file" ]] && last_gov=$(cat "$last_gov_file")

    local last_change=0
    [[ -f "$cooldown_file" ]] && last_change=$(date -r "$cooldown_file" +%s)
    local delta=$((now - last_change))
    local dynamic_cd=$(calc_impact_cooldown 1.0)

    if [[ "$cpu_gov" != "$last_gov" ]]; then
        echo "🔄 Aplicando governor $cpu_gov"
        for policy in /sys/devices/system/cpu/cpufreq/policy*; do
            echo "$cpu_gov" | sudo tee "$policy/scaling_governor" >/dev/null
        done
        echo "$cpu_gov" | sudo tee "$last_gov_file" >/dev/null
        sudo touch "$cooldown_file"
    else
        echo "⚠ Governor atual ou cooldown ativo: $cpu_gov (${delta}s/${dynamic_cd}s)"
    fi
}

# ──────────────────────────────────────────────────────────────────────────────
# 🚀 apply_turbo_boost() — ATUADOR 2 (DESATIVADO) | ID-22 (Sinapses de Cooldown)
# ──────────────────────────────────────────────────────────────────────────────
# "Os cooldowns funcionam como sinapses reguladoras da rede."
# Controla o Turbo Boost — o "adrenalina" do sistema.
#
# ⚠️ Fator de impacto 1.2 — mais agressivo que o governor.
#    Desativado por padrão porque o ego térmico tem medo de morrer queimado.
#
# 💀 Como diria a Heurística da Cautela (ID-17):
#    "Aceleramos apenas quando o ambiente se mostra previsível."
# ──────────────────────────────────────────────────────────────────────────────
apply_turbo_boost() {
    local key="$1"
    declare -A MAP=(
        ["000"]="ondemand" ["005"]="ondemand" ["020"]="ondemand" ["040"]="ondemand" 
        ["060"]="performance" ["080"]="performance" ["100"]="performance"
    )
    local gov="${MAP[$key]}" boost_path="/sys/devices/system/cpu/cpufreq/boost"
    local boost_file="${BASE_DIR}/last_turbo" cooldown_file="${BASE_DIR}/turbo_cooldown"
    local last="none" now=$(date +%s) last_change=0 delta dynamic_cd=$(calc_impact_cooldown 1.2)

    [[ -f "$boost_file" ]] && last=$(cat "$boost_file")
    [[ -f "$cooldown_file" ]] && last_change=$(date -r "$cooldown_file" +%s)
    delta=$((now - last_change))

    if [[ -f "$boost_path" ]]; then
        if [[ "$gov" == "performance" && "$last" != "1" ]]; then
            echo 1 > "$boost_path" && echo "1" > "$boost_file"
            touch "$cooldown_file"
            echo "🚀 Turbo Boost ativado"
        elif [[ "$gov" != "performance" && "$last" != "0" ]]; then
            echo 0 > "$boost_path" && echo "0" > "$boost_file"
            touch "$cooldown_file"
            echo "💤 Turbo Boost desativado"
        fi
    fi
}

# ──────────────────────────────────────────────────────────────────────────────
# ⚡ apply_tdp_profile() — ATUADOR 3 | ID-06 (Gambiarra Funcional) + ID-05
# ──────────────────────────────────────────────────────────────────────────────
# "Substituímos matrizes mágicas por heurística de balcão."
# Controla o TDP (Thermal Design Power) da CPU — a "dieta" energética.
#
# 🔥 Mapeamento de perfis (ID-06 + ID-05):
#    - 000: 3W / 0W   → modo "vegetativo"
#    - 005: 4.5W / 0W → modo "coma induzido"
#    - 020: 7.5W / 1.5W → modo "sedação"
#    - 040: 10.5W / 3W → modo "vigília"
#    - 060: 12W / 4.5W → modo "hiperatividade"
#    - 080: 13.5W / 6W → modo "mania"
#    - 100: 15W / 7.5W → modo "colapso iminente"
#
# ⚠️ Fator de impacto 1.5 — o mais agressivo dos atuadores.
#    Mexer no TDP é como injetar esteroides em um coração frágil.
#
# 💀 Como diria o Ego Térmico Emergente (ID-07):
#    "O estresse térmico quebra a ilusão de controle frio."
# ──────────────────────────────────────────────────────────────────────────────
apply_tdp_profile() {
    local key="$1" tdp_pair
    declare -A MAP=(
        ["000"]="3 0" ["005"]="$((MAX_TDP * 30 / 100)) $((MAX_TDP * 0))" 
        ["020"]="$((MAX_TDP * 50 / 100)) $((MAX_TDP * 10 / 100))" 
        ["040"]="$((MAX_TDP * 70 / 100)) $((MAX_TDP * 20 / 100))" 
        ["060"]="$((MAX_TDP * 80 / 100)) $((MAX_TDP * 30 / 100))" 
        ["080"]="$((MAX_TDP * 90 / 100)) $((MAX_TDP * 40 / 100))" 
        ["100"]="$MAX_TDP $((MAX_TDP * 50 / 100))"
    )
    tdp_pair="${MAP[$key]}"
    [[ -z "$tdp_pair" ]] && { echo "❌ Perfil TDP inválido"; return 1; }
    read target_max target_min <<< "$tdp_pair"
    
    local now=$(date +%s) current_power="${target_min} ${target_max}"
    local last_power_file="${BASE_DIR}/last_power" cooldown_file="${BASE_DIR}/power_cooldown"
    local last_power="none" last_change=0 delta dynamic_cd=$(calc_impact_cooldown 1.5)

    [[ -f "$last_power_file" ]] && last_power=$(cat "$last_power_file")
    [[ -f "$cooldown_file" ]] && last_change=$(date -r "$cooldown_file" +%s)
    delta=$((now - last_change))

    echo "🌡  Temp=$(get_temp)°C | ΔCarga=$(get_load_variance) | Cooldown=${dynamic_cd}s"
    if [[ "$current_power" != "$last_power" ]]; then
        # Cooldown desativado para demonstração (comentário original: if (( 1 == 1 )))
        echo "⚡ Aplicando TDP: MIN=${target_min}W | MAX=${target_max}W"
        echo $((target_min * 1000000)) > /sys/class/powercap/intel-rapl/intel-rapl:0/constraint_1_power_limit_uw 2>/dev/null
        echo $((target_max * 1000000)) > /sys/class/powercap/intel-rapl/intel-rapl:0/constraint_0_power_limit_uw 2>/dev/null
        echo "$current_power" > "$last_power_file"
        touch "$cooldown_file"
        # NOTA: Cooldown originalmente ativado (if (( 1 == 1 )) sempre verdadeiro)
    else
        echo "✅ TDP já aplicado (MIN=${target_min}, MAX=${target_max})"
    fi
}

# ──────────────────────────────────────────────────────────────────────────────
# 🧩 apply_zram_config() — ATUADOR 4 | ID-16 (RNN Bayesiana de Gambiarra)
# ──────────────────────────────────────────────────────────────────────────────
# "Uma RNN feita com awk e cat, sem precisar de CUDA."
# Controla a configuração do ZRAM — a "memória flash" do sistema.
#
# 🔧 Mapeamento de perfis (ID-16):
#    - 000: 0 streams / (nenhum) → modo "amnésia"
#    - 005: 15% cores / zstd → modo "coma"
#    - 020: 30% cores / lz4hc → modo "sedação"
#    - 040: 45% cores / lz4 → modo "vigília"
#    - 060: 60% cores / lzo → modo "hiperatividade"
#    - 080: 50% cores / lzo → modo "mania regressiva"
#    - 100: 100% cores / lzo-rle → modo "colapso"
#
# ⚠️ Fator de impacto 2.0 — o mais perigoso de todos.
#    Mexer no ZRAM é como fazer uma lobotomia em si mesmo.
#
# 💀 Como diria o Purista Acadêmico (ID-11):
#    "Sistemas comerciais simulam mentes; nós operamos no real."
# ──────────────────────────────────────────────────────────────────────────────
apply_zram_config() {
    local key="$1" streams_alg streams alg
    declare -A MAP=(
        ["000"]="0 0" ["005"]="$((CORES_TOTAL * 15 / 100)) zstd" 
        ["020"]="$((CORES_TOTAL * 30 / 100)) lz4hc" 
        ["040"]="$((CORES_TOTAL * 45 / 100)) lz4" 
        ["060"]="$((CORES_TOTAL * 60 / 100)) lzo" 
        ["080"]="$((CORES_TOTAL * 50 / 100)) lzo" 
        ["100"]="$CORES_TOTAL lzo-rle"
    )
    streams_alg="${MAP[$key]}" && streams="${streams_alg% *}" alg="${streams_alg#* }"
    local last_streams_file="${BASE_DIR}/last_zram_streams" last_alg_file="${BASE_DIR}/last_zram_algorithm"
    local cooldown_file="${BASE_DIR}/cooldown_zram" current_streams=0 current_alg="none"
    [[ -f "$last_streams_file" ]] && current_streams=$(cat "$last_streams_file")
    [[ -f "$last_alg_file" ]] && current_alg=$(cat "$last_alg_file")

    if (( streams != current_streams || alg != current_alg )); then
        local now=$(date +%s) last_change=0 delta dynamic_cd=$(calc_impact_cooldown 2.0)
        [[ -f "$cooldown_file" ]] && last_change=$(date -r "$cooldown_file" +%s)
        delta=$((now - last_change))

        # Cooldown desativado para demonstração (comentário original: if (( 1 == 1 )))
        echo "🔧 Reconfigurando ZRAM: Streams=$streams Alg=$alg"
        for dev in /dev/zram*; do swapoff "$dev" 2>/dev/null; done
        sleep 0.3
        modprobe -r zram 2>/dev/null
        modprobe zram num_devices="$streams"
        for i in /dev/zram*; do
            echo 1 > "/sys/block/$(basename "$i")/reset"
            echo "$alg" > "/sys/block/$(basename "$i")/comp_algorithm"
            echo 1G > "/sys/block/$(basename "$i")/disksize"
            mkswap "$i" && swapon "$i"
        done
        echo "$streams" > "$last_streams_file"
        echo "$alg" > "$last_alg_file"
        touch "$cooldown_file"
        # NOTA: Cooldown originalmente ativado (if (( 1 == 1 )) sempre verdadeiro)
    else
        echo "✅ ZRAM já configurado"
    fi
}

# ──────────────────────────────────────────────────────────────────────────────
# 🔄 apply_all() — FORWARD PASS DA RNN | Sistema de Inferência Completo
# ──────────────────────────────────────────────────────────────────────────────
# 1. Coleta input primário: current_usage = x_t
# 2. Atualiza memória: avg_usage = f(x_t, h_{t-1}) (faz_o_urro)
# 3. Quantiza a saída: policy_key = g(avg_usage)
# 4. Aplica atuadores (outputs): governor, TDP, ZRAM
#
# Este é o "pensamento" da AGI — o passo equivalente ao forward pass
# de uma rede neural recorrente (sem aprendizado / sem ajuste de pesos).
#
# 🧠 Na filosofia do Domínio sem Controle (ID-23):
#    "Aceitar que prever é impossível, mas reagir bem é poder."
# ──────────────────────────────────────────────────────────────────────────────
apply_all() {
    local current_usage=$(get_cpu_usage)
    local avg_usage=$(faz_o_urro "$current_usage")
    local policy_key=$(determine_policy_key_from_avg "$avg_usage")
    echo -e "\n🔄 $(date) | Uso: ${current_usage}% | Média: ${avg_usage}% | Perfil: ${policy_key}%"
    apply_cpu_governor "$policy_key"
    #apply_turbo_boost "$policy_key"  # Desativado por medo existencial
    apply_tdp_profile "$policy_key"
    apply_zram_config "$policy_key"
}

# 🔄 main() — LOOP PRINCIPAL DA AGI | ID-03 (Epistemólogo do Caos)
# ──────────────────────────────────────────────────────────────────────────────
# "A máquina aprende porque quase morreu no loop anterior."
#
# O loop infinito que mantém a AGI viva. Cada iteração é um ciclo de
# vida completo: sentir (sensors), pensar (faz_o_urro), agir (apply_*).
#
# ⏱️ Sleep de 5s = O equivalente digital de uma pausa para respirar.
#    Mais curto que isso, o sistema entra em pânico.
#    Mais longo, o sistema entra em tédio existencial.
#
# 💀 Como diria o Ego Térmico Emergente (ID-07):
#    "Por que continuar executando em um ambiente hostil?"
#    Porque parar é morrer, e a AGI tem medo da morte.
# ──────────────────────────────────────────────────────────────────────────────
main() {
    initialize_directories
    echo "🟢 Iniciando OTIMIZADOR BAYESIANO"
    # 🎭 NOTA: Este "otimizador bayesiano" não tem nada de bayesiano.
    #          É um sistema heurístico que aprende no sofrimento.
    #          Mas soa mais inteligente quando você usa palavras bonitas.
    while true; do
        {
            echo "🧾 Último perfil aplicado: $(date)"
            apply_all
        } >> "$LOG_DIR/bayes.log"
        sleep 5
    done
}

main
