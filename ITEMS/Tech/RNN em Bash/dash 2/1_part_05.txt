void OnTick()
{
    StartNewCycle();

    // 1. Atualizar Dados
    if(!symbolInfo.RefreshRates()) {
        LogBool("OnTick", "RefreshRates", false);
        return;
    }
    LogBool("OnTick", "RefreshRates", true);
   
    // 2. Análise Macro (Mesh + Fibo)
    datetime currentTime[1];
    if(CopyTime(_Symbol, _Period, 0, 1, currentTime) > 0) {
        if(currentTime[0] != lastBarTime) {
            lastBarTime = currentTime[0];
            UpdateMacroAnalysis();
        }
    }
   
    // 3. Verificar Hard Stop Loss
    CheckHardStopLoss();
   
    // 4. Gerenciar Trailing Stop
    ManageTrailingStop();
   
    // 5. Verificar Sinais de Entrada
    
    // REGRA 1: Verificar se este símbolo já tem posição (Limite: 1 por ativo)
    bool symbolBusy = IsSymbolBusy();
    LogBool("OnTick", "IsSymbolBusy", symbolBusy);
    if(symbolBusy) {
        LogString("OnTick", "BlockReason", "Symbol já tem posição");
        PrintCycleLog();
        return;
    }

    // REGRA 2: Verificar Limite Global (Limite: 5 por conta)
    int globalPositions = CountGlobalPositions();
    LogValue("OnTick", "GlobalPositions", (double)globalPositions);
    if(globalPositions >= MaxOpenPositions) {
        LogString("OnTick", "BlockReason", "Limite global atingido");
        PrintCycleLog();
        return;
    }

    // Atraso aleatório pequeno para evitar conflito de execução
    int randomDelay = MathRand() % 100;
    LogValue("OnTick", "RandomDelay", (double)randomDelay);
    Sleep(randomDelay); 
    
    CheckEntrySignalsWithScore();
    
    // Imprimir log do ciclo
    PrintCycleLog();
}

//| Verifica se JÁ existe ordem neste símbolo (CORREÇÃO FUNDAMENTAL)|

bool IsSymbolBusy()
{
    int positionsChecked = 0;
    int ordersChecked = 0;
    
    // Verifica Posições Abertas
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        positionsChecked++;
        if(PositionGetTicket(i) > 0) {
            // Verifica se é nossa (MagicNumber) E se é deste símbolo
            if(PositionGetInteger(POSITION_MAGIC) == currentMagicNumber && 
               PositionGetString(POSITION_SYMBOL) == _Symbol) {
                LogValue("IsSymbolBusy", "PositionsChecked", (double)positionsChecked);
                LogValue("IsSymbolBusy", "OrdersChecked", (double)ordersChecked);
                return true; // Já tem posição, bloqueia nova entrada
            }
        }
    }
    
    // Verifica Ordens Pendentes (para não duplicar na execução)
    for(int i = OrdersTotal() - 1; i >= 0; i--) {
        ordersChecked++;
        if(OrderGetTicket(i) > 0) {
            if(OrderGetInteger(ORDER_MAGIC) == currentMagicNumber && 
               OrderGetString(ORDER_SYMBOL) == _Symbol) {
                LogValue("IsSymbolBusy", "PositionsChecked", (double)positionsChecked);
                LogValue("IsSymbolBusy", "OrdersChecked", (double)ordersChecked);
                return true; // Já tem ordem pendente
            }
        }
    }
    
    LogValue("IsSymbolBusy", "PositionsChecked", (double)positionsChecked);
    LogValue("IsSymbolBusy", "OrdersChecked", (double)ordersChecked);
    return false;
}

    LogValue("CountGlobalPositions", "OrdersCounted", (double)ordersCounted);
    LogValue("CountGlobalPositions", "TotalCount", (double)count);

            points += 0.30; 
            reasons += "EMA+ "; 
            LogBool("CalculateSignalScore", "EMA_SellSignal", true);
        }
    }

    if(isBuySignal) {

            reasons += "RSI_Over+ ";

void CheckEntrySignalsWithScore()
{
    double zigZagBuffer[], high[], low[];
    datetime time[];
    ArraySetAsSeries(zigZagBuffer, true);
    ArraySetAsSeries(high, true);
    ArraySetAsSeries(low, true);
    ArraySetAsSeries(time, true);
   
    int checkRange = 10;
    LogValue("CheckEntrySignalsWithScore", "CheckRange", (double)checkRange);
    
    if(CopyBuffer(handleZigZag, 0, 0, checkRange, zigZagBuffer) < checkRange) {
        LogString("CheckEntrySignalsWithScore", "Error", "Falha copiar ZigZag");
        return;
    }
    if(CopyHigh(_Symbol, _Period, 0, checkRange, high) < checkRange) {
        LogString("CheckEntrySignalsWithScore", "Error", "Falha copiar High");
        return;
    }
    if(CopyLow(_Symbol, _Period, 0, checkRange, low) < checkRange) {
        LogString("CheckEntrySignalsWithScore", "Error", "Falha copiar Low");
        return;
    }
    if(CopyTime(_Symbol, _Period, 0, checkRange, time) < checkRange) {
        LogString("CheckEntrySignalsWithScore", "Error", "Falha copiar Time");
        return;
    }
    
    LogBool("CheckEntrySignalsWithScore", "DataCopied", true);
   
    for(int i = 1; i < checkRange; i++) {
        LogValue("CheckEntrySignalsWithScore", "CheckingIndex", (double)i);
        
        double zVal = zigZagBuffer[i];
        LogValue("CheckEntrySignalsWithScore", "ZigZagValue", zVal);
        
        if(zVal != 0.0 && zVal != EMPTY_VALUE) {
            if(time[i] <= lastTradeTime) {
                LogString("CheckEntrySignalsWithScore", "SkipReason", "Time <= lastTradeTime");
                continue;
            }
           
            bool isHigh = (MathAbs(high[i] - zVal) < _Point);
            bool isLow = (MathAbs(low[i] - zVal) < _Point);
            
            LogBool("CheckEntrySignalsWithScore", "IsHigh", isHigh);
            LogBool("CheckEntrySignalsWithScore", "IsLow", isLow);
            LogValue("CheckEntrySignalsWithScore", "High[i]", high[i]);
            LogValue("CheckEntrySignalsWithScore", "Low[i]", low[i]);
           
            if(isHigh || isLow) {
                bool isBuySignal = isLow;
                LogBool("CheckEntrySignalsWithScore", "IsBuySignal", isBuySignal);
                
                SignalScore score = CalculateSignalScore(isBuySignal, zVal);
                
                if(score.isValid) {
                    LogString("CheckEntrySignalsWithScore", "SignalValid", "TRUE");
                    
                    // --- VERIFICAÇÃO FINAL ANTES DE ENVIAR A ORDEM ---
                    if(IsSymbolBusy()) {
                        LogString("CheckEntrySignalsWithScore", "Blocked", "SymbolBusy");
                        return; // 1 por Ativo
                    }
                    if(CountGlobalPositions() >= MaxOpenPositions) {
                        LogString("CheckEntrySignalsWithScore", "Blocked", "MaxPositions");
                        return; // 5 Global
                    }

                    double lotSize = GetLotSizeByConfidence(score.confidence);
                    
                    LogValue("CheckEntrySignalsWithScore", "LotSize", lotSize);
                    LogValue("CheckEntrySignalsWithScore", "SignalConfidence", score.confidence);
                    
                    Print("Sinal Detectado: ", _Symbol, " | Conf: ", score.confidence);
                    DrawSignalArrow(isBuySignal, zVal, time[i]);
                    
                    if(InpEnableTrading) {
                        ExecuteTrade(isBuySignal, lotSize, score.confidence);
                    }
                    
                    lastTradeTime = time[i];
                    LogValue("CheckEntrySignalsWithScore", "LastTradeTime", (double)lastTradeTime);
                    break;
                } else {
                    LogString("CheckEntrySignalsWithScore", "SignalValid", "FALSE");
                }
            }
        }
    }
}

//+------------------------------------------------------------------+
//| Executar Trade                                                   |
//+------------------------------------------------------------------+
void ExecuteTrade(bool isBuy, double lotSize, double confidence)
{
    LogString("ExecuteTrade", "Starting", "Iniciando execução de trade");
    LogBool("ExecuteTrade", "IsBuy", isBuy);
    LogValue("ExecuteTrade", "LotSize", lotSize);
    LogValue("ExecuteTrade", "Confidence", confidence);
    
    double entry = isBuy ? symbolInfo.Ask() : symbolInfo.Bid();
    double slPoints = InitialStopLoss * 10 * _Point;
    double tpPoints = InitialTakeProfit * 10 * _Point;
    
    LogValue("ExecuteTrade", "EntryPrice", entry);
    LogValue("ExecuteTrade", "SL_Points", slPoints);
    LogValue("ExecuteTrade", "TP_Points", tpPoints);
   
    double sl = isBuy ? NormalizeDouble(entry - slPoints, _Digits) : NormalizeDouble(entry + slPoints, _Digits);
    double tp = isBuy ? NormalizeDouble(entry + tpPoints, _Digits) : NormalizeDouble(entry - tpPoints, _Digits);
    
    LogValue("ExecuteTrade", "StopLoss", sl);
    LogValue("ExecuteTrade", "TakeProfit", tp);
    
    string comment = StringFormat("V3_C%.0f", confidence * 100);
    LogString("ExecuteTrade", "Comment", comment);
   
    bool res = false;
    if(isBuy) {
        res = trade.Buy(lotSize, _Symbol, entry, sl, tp, comment);
        LogString("ExecuteTrade", "OrderType", "BUY");
    }
    else {
        res = trade.Sell(lotSize, _Symbol, entry, sl, tp, comment);
        LogString("ExecuteTrade", "OrderType", "SELL");
    }
    
    if(res) {
        Print("Ordem enviada com sucesso: ", _Symbol);
        LogBool("ExecuteTrade", "OrderSuccess", true);
        LogString("ExecuteTrade", "Result", "Sucesso");
    }
    else {
        Print("Erro ao enviar ordem: ", trade.ResultRetcodeDescription());
        LogBool("ExecuteTrade", "OrderSuccess", false);
        LogString("ExecuteTrade", "Result", trade.ResultRetcodeDescription());
    }
}

//+------------------------------------------------------------------+
//| Gerenciar Trailing Stop                                          |
//+------------------------------------------------------------------+
void ManageTrailingStop()
{
    int positionsChecked = 0;
    int trailingAdjusted = 0;
    
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        positionsChecked++;
        ulong ticket = PositionGetTicket(i);
        if(ticket > 0 && PositionSelectByTicket(ticket)) {
            
            // Só gerencia o que é nosso
            if(PositionGetInteger(POSITION_MAGIC) == currentMagicNumber) {
                
                // Trailing também só deve atuar no símbolo atual do gráfico para evitar conflitos de cache
                if(PositionGetString(POSITION_SYMBOL) == _Symbol) {
                    
                    double profit = PositionGetDouble(POSITION_PROFIT);
                    LogValue("ManageTrailingStop", "Ticket", (double)ticket);
                    LogValue("ManageTrailingStop", "Profit", profit);
                    
                    if(profit >= TrailingActivation) {
                        LogBool("ManageTrailingStop", "TrailingActive", true);
                        
                        double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
                        double stopLoss = PositionGetDouble(POSITION_SL);
                        double takeProfit = PositionGetDouble(POSITION_TP);
                        double trailPts = TrailingDistance * 10 * _Point;
                        
                        LogValue("ManageTrailingStop", "CurrentPrice", currentPrice);
                        LogValue("ManageTrailingStop", "CurrentSL", stopLoss);
                        LogValue("ManageTrailingStop", "TrailPoints", trailPts);
                        
                        double newSL = 0;
                        bool update = false;

                        if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY) {
                            newSL = NormalizeDouble(currentPrice - trailPts, _Digits);
                            if(newSL > stopLoss || stopLoss == 0) update = true;
                            LogString("ManageTrailingStop", "PositionType", "BUY");
                        } else {
                            newSL = NormalizeDouble(currentPrice + trailPts, _Digits);
                            if(newSL < stopLoss || stopLoss == 0) update = true;
                            LogString("ManageTrailingStop", "PositionType", "SELL");
                        }
                        
                        LogValue("ManageTrailingStop", "NewSL", newSL);
                        LogBool("ManageTrailingStop", "ShouldUpdate", update);
                        
                        if(update) {
                            trade.PositionModify(ticket, newSL, takeProfit);
                            trailingAdjusted++;
                            LogBool("ManageTrailingStop", "Adjusted", true);
                        }
                    } else {
                        LogBool("ManageTrailingStop", "TrailingActive", false);
                    }
                }
            }
        }
    }
    
    LogValue("ManageTrailingStop", "PositionsChecked", (double)positionsChecked);
    LogValue("ManageTrailingStop", "TrailingAdjusted", (double)trailingAdjusted);
}

    }

}
