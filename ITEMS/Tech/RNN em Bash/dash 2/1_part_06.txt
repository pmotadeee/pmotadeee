O ZigZag é usado para identificar pontos de virada (máximas e mínimas) no preço. Podemos substituí-lo por uma lógica mais direta, como:

Fractais de Bill Williams:

Um fractal de alta é uma barra com o máximo mais alto que um certo número de barras antes e depois.

Um fractal de baixa é uma barra com o mínimo mais baixo que um certo número de barras antes e depois.

Máximas e Mínimas Locais:

Percorrer um buffer de preços (high e low) e identificar pontos onde o preço é o mais alto/baixo em uma janela de N barras.

Vantagens de uma abordagem mais simples:
Menos dependência de indicadores externos (ZigZag é um indicador personalizado que pode não estar disponível em todas as plataformas).

Controle total sobre a lógica de detecção.

Potencialmente mais rápido, pois evita a inicialização de um indicador separado.

Camada de Percepção:

Sugestão de Implementação:

Podemos criar uma função que, dado um array de highs e lows, retorna arrays de máximas e mínimas locais. Parâmetros como o número de barras para cada lado (janela) podem ser ajustáveis.

Exemplo de detecção de fractal de alta (máxima local):

Para uma barra i, ela é uma máxima local se high[i] é o maior valor entre high[i-2], high[i-1], high[i], high[i+1], high[i+2] (janela de 5 barras, com 2 de cada lado).

bool IsHighFractal(const double &high[], int index, int window=2) {
    // Verifica se a barra no índice 'index' é uma máxima local.
    // 'window' é o número de barras para cada lado.
    if (index < window || index >= ArraySize(high) - window) {
        return false; // Não há barras suficientes para comparar.
    }
    
    double currentHigh = high[index];
    for (int j = index - window; j <= index + window; j++) {
        if (j == index) continue;
        if (currentHigh <= high[j]) {
            return false;
        }
    }
    return true;
}

bool IsLowFractal(const double &low[], int index, int window=2) {
    // Verifica se a barra no índice 'index' é uma mínima local.
    if (index < window || index >= ArraySize(low) - window) {
        return false;
    }
    
    double currentLow = low[index];
    for (int j = index - window; j <= index + window; j++) {
        if (j == index) continue;
        if (currentLow >= low[j]) {
            return false;
        }
    }
    return true;
}

struct SmartPivot {
    double price;
    datetime time;
    int barIndex;
    bool isHigh;
    double confidence; // 0-1: confiança no pivô
    int fractalType;   // 1: Fractal simples, 2: Fractal estendido, 3: Maximum local
    bool systemConfirm; // Confirmado por outras camadas
};

// 1. DETECÇÃO BRUTA (Percepção)
SmartPivot[] DetectRawPivots(const double &high[], const double &low[], 
                             const datetime &time[], int lookback) {
    SmartPivot pivots[];
    
    for(int i = 3; i < lookback-3; i++) {
        // TIPO 1: Fractal de Bill Williams (5 barras)
        bool fractalHigh = IsFractalHigh(high, i);
        bool fractalLow = IsFractalLow(low, i);
        
        // TIPO 2: Maximum/Minimum Local (7 barras - mais robusto)
        bool localHigh = IsLocalMaximum(high, i, 3); // 3 barras cada lado
        bool localLow = IsLocalMinimum(low, i, 3);
        
        // TIPO 3: Break de estrutura (mudança de tendência)
        bool structureBreak = CheckStructureBreak(high, low, i);
        
        if(fractalHigh || localHigh || structureBreak) {
            SmartPivot p;
            p.price = high[i];
            p.time = time[i];
            p.barIndex = i;
            p.isHigh = true;
            p.confidence = 0.3; // Base
            p.fractalType = fractalHigh ? 1 : (localHigh ? 2 : 3);
            p.systemConfirm = false;
            ArrayPush(pivots, p);
        }
        
        // Similar para lows...
    }
    return pivots;
}

// 2. VALIDAÇÃO CONTEXTUAL (Inferência)
void ValidatePivotsWithContext(SmartPivot &pivots[], const double &emaFast[],
                               const double &emaSlow[], const double &rsi[]) {
    for(int i = 0; i < ArraySize(pivots); i++) {
        double contextScore = 0.0;
        
        // Confirmação por tendência
        if(pivots[i].isHigh) {
            if(emaFast[pivots[i].barIndex] > emaSlow[pivots[i].barIndex]) {
                contextScore += 0.2; // Alta em tendência de alta
            }
        } else {
            if(emaFast[pivots[i].barIndex] < emaSlow[pivots[i].barIndex]) {
                contextScore += 0.2; // Baixa em tendência de baixa
            }
        }
        
        // Confirmação por momentum
        if(IsStrongMomentum(pivots[i].barIndex)) {
            contextScore += 0.3;
        }
        
        // Confirmação por volume/RSI
        if(rsi[pivots[i].barIndex] > 70 && pivots[i].isHigh) {
            contextScore += 0.2; // Alta com sobrecompra
        }
        if(rsi[pivots[i].barIndex] < 30 && !pivots[i].isHigh) {
            contextScore += 0.2; // Baixa com sobrevenda
        }
        
        pivots[i].confidence += contextScore;
        pivots[i].systemConfirm = (pivots[i].confidence >= 0.6);
    }
}

// 3. FILTRAGEM FINAL (Eliminação de ruído)
SmartPivot[] FilterValidPivots(const SmartPivot &pivots[], double minConfidence = 0.6) {
    SmartPivot validPivots[];
    
    for(int i = 0; i < ArraySize(pivots); i++) {
        // Filtro 1: Confiança mínima
        if(pivots[i].confidence < minConfidence) continue;
        
        // Filtro 2: Distância mínima entre pivôs
        bool tooClose = false;
        for(int j = 0; j < ArraySize(validPivots); j++) {
            if(MathAbs(pivots[i].barIndex - validPivots[j].barIndex) < 5) {
                tooClose = true;
                // Mantém o pivô com maior confiança
                if(pivots[i].confidence > validPivots[j].confidence) {
                    validPivots[j] = pivots[i];
                }
                break;
            }
        }
        if(!tooClose) {
            ArrayPush(validPivots, pivots[i]);
        }
    }
    
    return validPivots;
}
