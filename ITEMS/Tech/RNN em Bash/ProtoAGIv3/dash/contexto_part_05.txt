void ExecuteTrade(bool isBuy, double lotSize, double confidence)
{

    // 1. Obter Volatilidade (ATR) para definir SL Técnico
    double atrVal[1];
    if(CopyBuffer(handleAtr, 0, 0, 1, atrVal) < 1) return;
    double volatility = atrVal[0];
    
    // 2. Calcular Distância do SL (Risco)
    double slDistance = volatility * InpSlAtrMult;
    
    // Garantia mínima de SL (caso ATR seja muito baixo em calmaria)
    if(slDistance < 50 * _Point) slDistance = 50 * _Point;

    // 3. Calcular Distância do TP (Retorno) baseado no R:R
    double rrRatio = GetRiskRewardMultiplier(confidence);
    double tpDistance = slDistance * rrRatio;
    
    // 4. Definir Preços
    double sl = isBuy ? NormalizeDouble(entry - slDistance, _Digits) : NormalizeDouble(entry + slDistance, _Digits);
    double tp = isBuy ? NormalizeDouble(entry + tpDistance, _Digits) : NormalizeDouble(entry - tpDistance, _Digits);
    
    string comment = StringFormat("RR_1:%.0f_C%.0f", rrRatio, confidence * 100);
   
    bool res = false;
    if(isBuy) res = trade.Buy(lotSize, _Symbol, entry, sl, tp, comment);
    else res = trade.Sell(lotSize, _Symbol, entry, sl, tp, comment);
    
    if(res) {
        Print("ORDEM ABERTA | ", _Symbol, 
              " | Risco: ", DoubleToString(slDistance/_Point, 0), "pts",
              " | Retorno: ", DoubleToString(tpDistance/_Point, 0), "pts",
              " | R:R: 1:", DoubleToString(rrRatio, 1));
    } else {
        Print("Erro ao enviar ordem: ", trade.ResultRetcodeDescription());
    }
}

//| Gerenciar Trailing Stop Inteligente (R:R Awareness)              |
//+------------------------------------------------------------------+
void ManageSmartTrailing()
{
    for(int i = PositionsTotal() - 1; i >= 0; i--) {
        ulong ticket = PositionGetTicket(i);
        if(ticket > 0 && PositionSelectByTicket(ticket)) {
            
            if(PositionGetInteger(POSITION_MAGIC) == currentMagicNumber && 
               PositionGetString(POSITION_SYMBOL) == _Symbol) {
                
                double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
                double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
                double stopLoss = PositionGetDouble(POSITION_SL);
                double takeProfit = PositionGetDouble(POSITION_TP);
                long type = PositionGetInteger(POSITION_TYPE);
                
                // Pegar ATR atual para referência
                double atrVal[1]; 
                if(CopyBuffer(handleAtr, 0, 0, 1, atrVal) < 1) continue;
                
                double riskUnit = atrVal[0] * InpSlAtrMult; // Unidade de Risco (1R)
                if(riskUnit <= 0) riskUnit = 100 * _Point; // Fallback
                
                double profitPoints = 0;
                if(type == POSITION_TYPE_BUY) profitPoints = currentPrice - openPrice;
                else profitPoints = openPrice - currentPrice;
                
                // --- FASE 1: BREAKEVEN (Proteção Rápida) ---
                if(UseBreakeven && profitPoints >= (riskUnit * BreakevenTriggerRR)) {
                    double beLevel = 0;
                    if(type == POSITION_TYPE_BUY) {
                        beLevel = openPrice + (10 * _Point); // Open + Spread leve
                        if(stopLoss < openPrice) trade.PositionModify(ticket, beLevel, takeProfit);
                    } else {
                        beLevel = openPrice - (10 * _Point);
                        if(stopLoss > openPrice || stopLoss == 0) trade.PositionModify(ticket, beLevel, takeProfit);
                    }
                }
                
                // --- FASE 2: TRAILING DE LUCRO (Maximizar Ganhos) ---
                if(profitPoints >= riskUnit) {
                    double trailDist = riskUnit * 0.8; // Mantém o stop a 0.8R de distância
                    double newSL = 0;
                    bool update = false;
                    
                    if(type == POSITION_TYPE_BUY) {
                        newSL = NormalizeDouble(currentPrice - trailDist, _Digits);
                        if(newSL > stopLoss) update = true;
                    } else {
                        newSL = NormalizeDouble(currentPrice + trailDist, _Digits);
                        if(newSL < stopLoss || stopLoss == 0) update = true;
                    }
                    
                    if(update) trade.PositionModify(ticket, newSL, takeProfit);
                }
            }
        }
    }
}

    double points = 0.0;
   
    // 1. EMAs (30%)
    if(isBuySignal) { if(emaFast[0] > emaSlow[0]) points += 0.30; }
    else { if(emaFast[0] < emaSlow[0]) points += 0.30; }
   
    // 2. RSI (25%)
    if(isBuySignal) { if(rsi[0] < 30) points += 0.25; else if(rsi[0] < 50) points += 0.15; }
    else { if(rsi[0] > 70) points += 0.25; else if(rsi[0] > 50) points += 0.15; }
   
    // 3. Momentum (20%)
    if(isBuySignal) { if(emaFast[0] > emaFast[1]) points += 0.20; }
    else { if(emaFast[0] < emaFast[1]) points += 0.20; }
   
    // 4. ZigZag (25%)
    points += 0.25; 
   
    score.confidence = points;
    score.isValid = (points >= 0.40); 
   
    return score;
}

    }

Vamos explicar tecnicamente por que o sistema é resiliente, baseado no código:

Gestão de Risco Agressiva:

Hard Stop Loss Global: O sistema monitora o prejuízo total de cada posição (incluindo swap e comissão) e fecha a posição se atingir um limite absoluto (HardStopLoss). Isso impede que uma posição perdedora drene o capital além de um ponto fixo.

Tamanho de Posição Baseado em Confiança: O sistema usa lotes pequenos e escalonados (0.05, 0.25, 0.50) com base na confiança do sinal. Isso significa que mesmo em sinais de alta confiança, o tamanho da posição é limitado, o que reduz o risco de perdas catastróficas.

Multiplicador de ATR para Stop Loss: O stop loss é definido com base na volatilidade atual (ATR). Isso ajusta o risco à condição do mercado, evitando stops muito apertados em mercados voláteis ou muito largos em mercados calmos.

Controles de Exposição:

Limite de Posições Globais: O sistema não permite mais de 30 posições abertas (MaxOpenPositions). Isso evita a sobrecarga da conta e do VPS, e também limita a exposição total.

Não permite múltiplas entradas no mesmo símbolo: A função IsSymbolBusy() verifica se já há uma posição ou ordem no símbolo atual. Isso evita acumular posições em uma única direção e ser pego em uma reversão.

Entrada Condicional com Filtros:

O sistema só entra em trades quando o sinal atinge um score mínimo (40%). Esse score é composto por múltiplos fatores (EMA, RSI, Momentum, ZigZag), o que significa que o sinal precisa ser confirmado por várias condições.

Além disso, o sistema espera um novo sinal (baseado no tempo do último trade) para evitar entrar em vários trades em rápida sucessão.

Trailing Stop Inteligente:

Breakeven: Assim que o trade atinge um lucro de 0.5 vezes o risco (BreakevenTriggerRR), o stop loss é movido para o preço de entrada (mais um pequeno spread). Isso protege o capital inicial.

Trailing Progressivo: Depois que o lucro atinge 1 vez o risco, o sistema começa a mover o stop loss para travar lucros, mantendo uma distância de 0.8 vezes o risco. Isso permite que o trade continue lucrativo enquanto protege os ganhos.

Simplicidade e Determinismo:

O sistema não usa machine learning ou modelos complexos. Ele é baseado em regras claras e indicadores técnicos que são calculados da mesma forma em qualquer condição. Isso elimina o risco de overfitting (que é comum em sistemas complexos) e torna o comportamento previsível e estável.

Proteção contra Falhas Operacionais:

O sistema verifica constantemente a conexão e atualiza os dados do símbolo.

Usa Sleep(MathRand() % 100) para evitar execuções simultâneas em múltiplos gráficos (uma forma básica de prevenir condições de corrida).

Adaptabilidade Limitada:

O sistema não tenta prever o futuro, mas reage aos sinais atuais. Ele é desenhado para ser robusto em diferentes condições de mercado porque não assume um regime específico. No entanto, é importante notar que ele pode ter períodos de perda em mercados laterais ou voláteis.

Conclusão Técnica:
A "imunidade a falhas" não é absoluta. O sistema pode e vai ter períodos de perda. No entanto, a combinação de gestão de risco conservadora, controles de exposição e mecanismos de proteção (como o Hard Stop Loss e o trailing stop) faz com que as perdas sejam limitadas e que o sistema sobreviva a longos períodos, mesmo com uma taxa de acerto moderada.

Em outras palavras, o sistema é projetado para minimizar o impacto das perdas e proteger os ganhos, o que é a base da sobrevivência no trading. A simplicidade do sistema também reduz os pontos de falha, tornando-o mais confiável em execução contínua.

No entanto, é crucial entender que nenhum sistema é perfeito. O mercado pode apresentar condições que gerem uma série de perdas consecutivas, e o sistema pode sofrer drawdowns. A gestão de capital e a limitação de risco são o que impedem que esses drawdowns destruam a conta.
