void OnTick()
{
    if(!symbolInfo.RefreshRates()) return;
    
    // Nova Vela
    datetime currentTime = iTime(_Symbol, PERIOD_CURRENT, 0);
    if(currentTime == lastBarTime) return;
    lastBarTime = currentTime;

    if(IsPositionOpen()) return;

    // --- PASSO 1: TENDÊNCIA ---
    int trendDirection = CheckTrendMesh();
    if(trendDirection == 0) return; 

    // --- PASSO 2: VSA ---
    int vsaSignal = GetVSASignal(1);
    if(vsaSignal == 0) return;
    
    if(trendDirection == 1 && vsaSignal != 1) return;
    if(trendDirection == -1 && vsaSignal != -1) return;

    // --- PASSO 3: RSI ---
    bool isBuy = (vsaSignal == 1);
    if(!CheckRSIFilter(isBuy)) return;

    // --- PASSO 4: SL ATR ---
    double atrValue = GetATRValue();
    if(atrValue == 0) return;
    
    double slDistance = atrValue * InpATR_Multiplier;
    double slPrice = 0.0, tpPrice = 0.0;
    double entryPrice = 0.0;
    
    if(isBuy) {
        entryPrice = symbolInfo.Ask();
        slPrice = entryPrice - slDistance;
        tpPrice = entryPrice + (slDistance * 1.5);
    } else {
        entryPrice = symbolInfo.Bid();
        slPrice = entryPrice + slDistance;
        tpPrice = entryPrice - (slDistance * 1.5);
    }
    
    slPrice = NormalizeDouble(slPrice, _Digits);
    tpPrice = NormalizeDouble(tpPrice, _Digits);
    double slPoints = slDistance / _Point;

    // --- PASSO 5: EXECUÇÃO COM VERIFICAÇÃO DE MARGEM ---
    // AQUI ESTAVA O ERRO DE COMPILAÇÃO: Atualizado para passar isBuy e entryPrice
    double lotSize = CalculateKellyLotSize(AccountInfoDouble(ACCOUNT_EQUITY), slPoints, isBuy, entryPrice);
    
    if(lotSize > 0) {
        if(isBuy) trade.Buy(lotSize, _Symbol, entryPrice, slPrice, tpPrice, "iMon V4.01 Trend");
        else trade.Sell(lotSize, _Symbol, entryPrice, slPrice, tpPrice, "iMon V4.01 Trend");
        
        if(trade.ResultRetcode() == TRADE_RETCODE_DONE) {
             Print("Ordem Executada. Lote: ", lotSize);
        } else {
             Print("Erro Trade: ", trade.ResultRetcodeDescription());
        }
    }
}

3. Por que isso é uma Rede Neural?
Analogia com um Perceptron:
Pesos e Bias: Os parâmetros dos indicadores (períodos das EMAs, RSI, ATR) atuam como pesos. O bias está nos limiares (ex: RSI_Max_Buy = 70).

Função de Ativação: Cada neurônio usa funções de ativação simples:

Tendência: degrau (1, 0, -1)

RSI: degrau (passa/não passa)

VSA: degrau (1, 0, -1)

Risco: linear (valor do ATR)

Aprendizado (Backpropagation Degenerada):
Não há backpropagation tradicional, mas o sistema "aprende" através do cálculo de Kelly baseado no histórico recente (últimos 50 trades).

O Kelly ajusta o tamanho da posição com base no win rate e no risk/reward, o que é uma forma de aprendizado por reforço simples.

4. A Filosofia da Simplicidade Algorítmica
Menos é Mais:
A rede tem apenas 7 entradas, 4 neurônios ocultos e 1 saída.

Cada neurônio é especializado em uma tarefa simples.

A combinação dessas decisões simples resulta em uma decisão complexa.

Robustez pela Simplicidade:
Por ter poucos parâmetros, o sistema é menos propenso a overfitting.

As regras são interpretáveis: cada condição tem um significado claro.

A adaptação ao mercado é feita pelo cálculo de Kelly, que ajusta o tamanho da posição, não as regras de entrada.

Evolução Degenerada:
O sistema não tenta prever o mercado, apenas reagir a condições específicas.

A "inteligência" está na seleção das condições (que são padrões clássicos de trading) e na gestão de risco.

def CONSENSUS_VALIDATOR(trend, vsa, rsi_ok, atr_risk):
    # Regra 1: Tendência e VSA devem concordar
    if trend != vsa: return False
    
    # Regra 2: RSI deve estar na zona segura
    if not rsi_ok: return False
    
    # Regra 3: Risco deve ser calculável
    if atr_risk <= 0: return False
    
    # Se passou todas as camadas:
    return {
        'direction': trend,
        'sl_distance': atr_risk,
        'confidence': 'HIGH_ENOUGH'
    }

def KELLY_EXECUTOR(signal, equity, sl_distance):
    # 1. Calcular histórico recente
    win_rate, avg_win, avg_loss = analyze_last_50_trades()
    
    # 2. Fórmula de Kelly degenerada
    kelly_raw = (win_rate * (avg_win/avg_loss + 1) - 1) / (avg_win/avg_loss)
    kelly_safe = kelly_raw * 0.25  # "Nunca aposte tudo"
    kelly_capped = min(kelly_safe, 0.02)  # "Máximo 2%"
    
    # 3. Converter para lotes
    risk_amount = equity * kelly_capped
    loss_per_lot = sl_distance * tick_value
    lots = risk_amount / loss_per_lot
    
    # 4. Verificar margem (neurônio de segurança)
    if not check_margin(lots): 
        reduce_lots_until_fits()
    
    return {
        'action': 'BUY' if signal > 0 else 'SELL',
        'lots': lots,
        'sl': calculate_sl(signal, sl_distance),
        'tp': calculate_tp(signal, sl_distance)
    }
Função de ativação: Fórmula matemática com múltiplos limitadores

def TENDENCIA_MESH(ema21, ema100, ema200):
    if ema21 > ema100 > ema200: return 1    # "Maré subindo forte"

def CONSENSUS_VALIDATOR(trend, vsa, rsi_ok, atr_risk):
    # Regra 1: Tendência e VSA devem concordar
    if trend != vsa: return False
    
    # Regra 2: RSI deve estar na zona segura
    if not rsi_ok: return False
    
    # Regra 3: Risco deve ser calculável
    if atr_risk <= 0: return False
    
    # Se passou todas as camadas:
    return {
        'direction': trend,
        'sl_distance': atr_risk,
        'confidence': 'HIGH_ENOUGH'
    }
Função de ativação: Portão lógico AND múltiplo
Peso filosófico: "Múltiplas confirmações ou nenhuma ação"

def KELLY_EXECUTOR(signal, equity, sl_distance):
    # 1. Calcular histórico recente
    win_rate, avg_win, avg_loss = analyze_last_50_trades()
    
    # 2. Fórmula de Kelly degenerada
    kelly_raw = (win_rate * (avg_win/avg_loss + 1) - 1) / (avg_win/avg_loss)
    kelly_safe = kelly_raw * 0.25  # "Nunca aposte tudo"
    kelly_capped = min(kelly_safe, 0.02)  # "Máximo 2%"
    
    # 3. Converter para lotes
    risk_amount = equity * kelly_capped
    loss_per_lot = sl_distance * tick_value
    lots = risk_amount / loss_per_lot
    
    # 4. Verificar margem (neurônio de segurança)
    if not check_margin(lots): 
        reduce_lots_until_fits()
    
    return {
        'action': 'BUY' if signal > 0 else 'SELL',
        'lots': lots,
        'sl': calculate_sl(signal, sl_distance),
        'tp': calculate_tp(signal, sl_distance)
    }
Função de ativação: Fórmula matemática com múltiplos limitadores
Peso filosófico: "Aposte proporcional à sua vantagem, nunca mais do que pode perder"

O código do iMon V4.01 não é uma rede neural no sentido tradicional, mas sim um conjunto de regras heurísticas que processam dados de mercado em camadas. No entanto, a discussão anterior elevou isso a uma "rede neural degenerada", que é uma metáfora para um sistema simples, mas eficaz, que toma decisões com base em múltiplas entradas e camadas de processamento.

Agora, o foco é em como o sistema lida com a previsão do futuro. Na verdade, o sistema não prevê o futuro no sentido de prever preços. Em vez disso, ele reage ao presente com base em regras que foram definidas para capturar oportunidades quando certas condições são atendidas. A previsão é implícita na suposição de que os padrões que funcionaram no passado (tendência, VSA, RSI, etc.) continuarão a funcionar no futuro.

O sistema é baseado na ideia de que o mercado exibe certas regularidades (como tendências, condições de sobrecompra/sobrevenda, anomalias de volume) e que, quando essas regularidades aparecem, há uma probabilidade maior de que o mercado se mova de uma certa maneira. No entanto, o sistema não tenta prever quanto tempo o movimento vai durar ou quão longe vai. Em vez disso, ele define um alvo (TP) com base no risco (SL) e em um múltiplo fixo (1.5x).

O cálculo do tamanho da posição usando Kelly modificado é uma forma de adaptar o risco com base no desempenho recente, o que é uma forma de aprendizado. No entanto, isso não é uma previsão, mas uma gestão de risco.

Para aprofundar, vamos considerar os seguintes pontos:

O agora (presente): O sistema coleta dados atuais do mercado (preços, indicadores) e os processa através de suas camadas (tendência, VSA, RSI, ATR). A saída é uma decisão binária: entrar ou não entrar em um trade, e se entrar, qual direção e tamanho.

O estado futuro: O sistema não modela o futuro explicitamente. Em vez disso, ele assume que, se as condições atuais atendem a critérios específicos, então o futuro próximo provavelmente será favorável. O take profit e stop loss são definidos com base na volatilidade atual (ATR) e em um múltiplo fixo, o que é uma forma de definir um alvo e um risco com base nas condições atuais.

Previsão implícita: A previsão é feita de forma implícita pela regra de entrada. Por exemplo, se a tendência é de alta e o VSA indica uma possível continuação, então o sistema "prevê" que o preço vai subir. No entanto, essa previsão não é quantificada em termos de probabilidade, mas sim transformada em uma decisão de trade.

Aprendizado: O sistema tem um componente de aprendizado apenas no tamanho da posição (Kelly). Ele não ajusta os critérios de entrada com base no passado. Portanto, a previsão (na forma de regras de entrada) é estática, enquanto a gestão de risco (tamanho da posição) é adaptativa.
