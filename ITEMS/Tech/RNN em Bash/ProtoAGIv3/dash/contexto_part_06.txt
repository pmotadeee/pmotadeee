      // Calcular novos níveis baseados no ATR
      double newSL = 0.0, newTP = 0.0;
      double slDistance = atrValue * InpSLATRMultiplier;
      double tpDistance = atrValue * InpTPATRMultiplier;
      
      // Garantir distâncias mínimas
      double minSLPoints = InpMinSLDistance * _Point;
      double minTPPoints = InpMinTPDistance * _Point;
      
      if(slDistance < minSLPoints) slDistance = minSLPoints;
      if(tpDistance < minTPPoints) tpDistance = minTPPoints;
      
      // Calcular novos preços
      if(type == POSITION_TYPE_BUY)
      {
         newSL = NormalizeDouble(currentPrice - slDistance, _Digits);
         newTP = NormalizeDouble(currentPrice + tpDistance, _Digits);
         
         // SL nunca pode subir acima do preço de entrada (proteção máxima)
         double entryPrice = PositionGetDouble(POSITION_PRICE_OPEN);
         if(newSL > entryPrice) newSL = entryPrice - (10 * _Point);

      else if(type == POSITION_TYPE_SELL)
      {
         newSL = NormalizeDouble(currentPrice + slDistance, _Digits);
         newTP = NormalizeDouble(currentPrice - tpDistance, _Digits);
         
         // SL nunca pode descer abaixo do preço de entrada
         double entryPrice = PositionGetDouble(POSITION_PRICE_OPEN);
         if(newSL < entryPrice) newSL = entryPrice + (10 * _Point);

      // Verificar se os níveis são válidos

      double minDist = (stopLevel + 10) * _Point;
      
      if(type == POSITION_TYPE_BUY)
      {
         if(currentPrice - newSL < minDist) newSL = currentPrice - minDist;
         if(newTP - currentPrice < minDist) newTP = currentPrice + minDist;
      }
      else
      {
         if(newSL - currentPrice < minDist) newSL = currentPrice + minDist;
         if(currentPrice - newTP < minDist) newTP = currentPrice - minDist;
      }
      
      // Verificar se há diferença significativa para atualizar
      double slDiff = MathAbs(newSL - currentSL);
      double tpDiff = MathAbs(newTP - currentTP);
      double minChange = 5 * _Point; // Mínimo 5 pontos de mudança

void OnTick() {
   if(!symbolInfo.RefreshRates()) return;
   
   // 1. Atualizar SL/TP dinamicamente a cada 1 segundo
   UpdateDynamicSLTP();
   
   // 2. Verificar stop loss diário
   if(!CheckDailyStopLoss()) return;
   
   // 3. Verificar limite de posições
   if(CountGlobalPositions() >= InpMaxOpenPositions) return;
   
   // 4. Verificar se já tem posição no símbolo
   if(IsSymbolPositionOpen()) return;
   
   // 5. Lógica de entrada (por barra)
   datetime currentTime = iTime(_Symbol, PERIOD_CURRENT, 0);
   if(currentTime == lastBarTime) return; 
   lastBarTime = currentTime;
   
   // 6. Verificar cooldown
   if(TimeCurrent() < lastTradeTime + (InpCooldownMinutes * 60)) return;
   
   // 7. Obter sinal VSA
   int vsaSignal = GetVSASignal(1); 
   if(vsaSignal == 0) return;
   
   double stopLossPrice = 0.0;
   double entryPrice = 0.0;
   bool isBuy = false;
   
   if(vsaSignal == 1) { 
      isBuy = true; 
      entryPrice = symbolInfo.Ask();
      stopLossPrice = GetNearestFractalLevel(false); 
      if(stopLossPrice >= entryPrice) return; 
   } else { 
      isBuy = false; 
      entryPrice = symbolInfo.Bid();
      stopLossPrice = GetNearestFractalLevel(true);
      if(stopLossPrice <= entryPrice) return;
   }
   
   // 8. Verificar distâncias mínimas
   long stopLevel = SymbolInfoInteger(_Symbol, SYMBOL_TRADE_STOPS_LEVEL);
   double minDist = (stopLevel + 20) * _Point;
   double safeDist = (double)SymbolInfoInteger(_Symbol, SYMBOL_SPREAD) * _Point + minDist;
   
   if(MathAbs(entryPrice - stopLossPrice) < safeDist) {
      if(isBuy) stopLossPrice = entryPrice - safeDist;
      else      stopLossPrice = entryPrice + safeDist;
   }
   
   // 9. Calcular score de confiança
   int confidence = CalculateConfidenceScore(isBuy, entryPrice, stopLossPrice);
   if(confidence < InpMinConfidence) {
      if(InpDebugMode) Print("Score: ", confidence, "% (Wait for 90%)");
      return;
   }
   
   // 10. Calcular tamanho do lote
   double lotSize = CalculateScoreLotSize(confidence);
   if(lotSize <= 0) return;
   
   // 11. Verificar margem
   double marginRequired = 0.0;
   ENUM_ORDER_TYPE oType = isBuy ? ORDER_TYPE_BUY : ORDER_TYPE_SELL;
   if(OrderCalcMargin(oType, _Symbol, lotSize, SymbolInfoDouble(_Symbol, SYMBOL_ASK), marginRequired)) {
      if(marginRequired > AccountInfoDouble(ACCOUNT_MARGIN_FREE)) {
         Print("Margem insuficiente. Abortar.");
         return; 
      }
   }
   
   // 12. Atualizar preços
   symbolInfo.RefreshRates();
   entryPrice = isBuy ? symbolInfo.Ask() : symbolInfo.Bid();
   
   if(isBuy) { 
      if(entryPrice - stopLossPrice < safeDist) stopLossPrice = entryPrice - safeDist; 
   } else { 
      if(stopLossPrice - entryPrice < safeDist) stopLossPrice = entryPrice + safeDist; 
   }
   
   // 13. Calcular Take Profit inicial (será ajustado dinamicamente depois)
   double tp = 0.0;
   double finalRiskDist = MathAbs(entryPrice - stopLossPrice);
   
   if(isBuy) tp = entryPrice + (finalRiskDist * 1.5);
   else      tp = entryPrice - (finalRiskDist * 1.5);
   
   // 14. Verificar se já tem posição (novamente por segurança)
   if(IsSymbolPositionOpen()) return;
   
   // 15. Executar ordem
   bool res = false;
   if(isBuy) {
      stopLossPrice = NormalizeDouble(stopLossPrice, _Digits);
      tp = NormalizeDouble(tp, _Digits);
      res = trade.Buy(lotSize, _Symbol, entryPrice, stopLossPrice, tp, "PaiMon V6 Score:"+IntegerToString(confidence));
   } else {
      stopLossPrice = NormalizeDouble(stopLossPrice, _Digits);
      tp = NormalizeDouble(tp, _Digits);
      res = trade.Sell(lotSize, _Symbol, entryPrice, stopLossPrice, tp, "PaiMon V6 Score:"+IntegerToString(confidence));
   }
   
   // 16. Registrar se sucesso
   if(res) {
      lastTradeTime = TimeCurrent();
      ulong ticket = trade.ResultOrder();
      AddPositionToTrack(ticket, entryPrice, stopLossPrice, tp);
      Print("PaiMon V6 SNIPER: Ordem Enviada. Score: ", confidence, "%");
   }
}

## **COMO FUNCIONA AGORA:**

### **A CADA 1 SEGUNDO (EXATAMENTE):**
1. Verifica todas as posições do robô (Magic Number)
2. Para cada posição:
   - Obtém ATR atual (volatilidade)
   - Calcula: `Novo SL = Preço atual ± (ATR × Multiplicador)`
   - Calcula: `Novo TP = Preço atual ± (ATR × Multiplicador)`
   - Garante distâncias mínimas
   - SL NUNCA piora a posição (só melhora)
   - Atualiza apenas se mudança >5 pontos

### **VANTAGENS:**
- **Adapta-se à volatilidade**: SL/TP expandem/contraem com o mercado
- **Proteção em tempo real**: Responde a movimentos bruscos
- **Minimiza chamadas**: Só atualiza se necessário (>5 pontos)
- **Não retrocede**: SL só melhora, nunca piora

### **EXEMPLO PRÁTICO:**
- **BTC em calmaria**: ATR = 50 pontos → SL = ±100 pts, TP = ±150 pts
- **BTC em volatilidade**: ATR = 200 pontos → SL = ±400 pts, TP = ±600 pts
- **Atualizações**: Apenas quando preço mudar significativamente

**Resultado:** SL/TP dinâmicos que se adaptam ao mercado em tempo real, protegendo lucros e limitando perdas de forma inteligente.

Níveis de SL/TP: baseados em fractais e ATR

CICLO DE FEEDBACK (APRENDIZADO)
Embora não tenha backpropagation tradicional, o sistema tem mecanismos de ajuste:

Ajuste de SL/TP dinâmico:

Baseado no ATR atual (volatilidade)

Atualizado a cada 1 segundo

Controle de risco:

Stop loss diário (InpMaxDailyLossPercent)

Cooldown entre trades (InpCooldownMinutes)

Limite de posições (InpMaxOpenPositions)

ANALOGIA COM UMA REDE NEURAL
Componente NN Tradicional	Equivalente no PaiMonInvoke
Input Layer	Dados de mercado + indicadores
Hidden Layers	Módulos VSA, EMA, RSI, Fractais
Activation Function	Funções de limiar (if-else)
Output Layer	Decisão de trade + parâmetros
Loss Function	Stop loss diário + hard stop
Backpropagation	Ajuste dinâmico de SL/TP
Dropout	Cooldown entre trades
ESTADOS DE AÇÃO (ACTION SPACE)
O sistema tem um espaço de ação discreto:

AÇÃO 0: Não operar (score < 90, cooldown, limite atingido)

AÇÃO 1: Comprar com lote X (0.01-0.20)

AÇÃO 2: Vender com lote X (0.01-0.20)

Parâmetros por ação:

Lote: função linear do score (90 → 0.01, 100 → 0.20)

SL: nível fractal mais próximo

TP: SL × 1.5 (inicial) + ajuste dinâmico

LIMITAÇÕES COMO REDE NEURAL
Sem aprendizado de pesos: Os pesos são fixos (20 pontos para tendência, etc.)

Sem generalização: Não treina com dados históricos

Arquitetura rígida: Não adapta a estrutura com o tempo

Feedback limitado: Apenas ajusta SL/TP, não os critérios de entrada

MELHORIAS POSSÍVEIS (PARA UMA VERDADEIRA REDE NEURAL)
Adicionar camada LSTM: Para memória de sequência temporal

Aprendizado por reforço: Usar P/L como função de recompensa

Autoajuste de pesos: Otimizar os pontos de cada critério

Redução de dimensionalidade: PCA nos indicadores de entrada

CONCLUSÃO
O PaiMonInvoke V6 é um sistema baseado em regras com pontuação, não uma rede neural verdadeira. No entanto, sua arquitetura em camadas (entrada → processamento → decisão) e mecanismo de feedback (ajuste dinâmico) o aproximam de uma rede neural simplificada. A principal diferença é a falta de aprendizado automático dos parâmetros - tudo é definido heuristicamente.

Para transformá-lo em uma rede neural real: Seria necessário substituir as funções de pontuação fixas por uma rede treinável (por exemplo, um MLP) que mapeie as entradas brutas diretamente para a decisão de trade, com os pesos aprendidos através de dados históricos.

def Risk_neuron(inputs):

    spread = inputs[5]       # Spread atual
